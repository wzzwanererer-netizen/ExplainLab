from importlib.metadata import version

for cmd in [["wsl.exe","--version"],["wsl.exe","--status"],["wsl.exe","-l","-v"],["wsl.exe","-e","uname","-a"],["sc.exe","query","WslService"],["sc.exe","query","LxssManager"]]:
    print("command",subprocess.list2cmdline(cmd),flush=True)
    try:
        p=subprocess.run(cmd,timeout=6,capture_output=True)
        for data in (p.stdout,p.stderr):
            if data:
                enc="utf-16-le" if b"\x00" in data else ("utf-8" if cmd[0]=="wsl.exe" else "mbcs")
                print(data.decode(enc,errors="replace").strip())
        print("exit_code",p.returncode)
    except (OSError,subprocess.TimeoutExpired) as exc:
        print(type(exc).__name__)
