 222: class ImportEnvelope(BaseModel):
 223:     """Active 导入审核 envelope（``payload.__a1_import``）。
 224: 
 225:     只承载 backend 在 approve 时需要的可重建上下文；Core ``_apply_update``
 226:     忽略本键，仅应用四字段。``context_digest`` 仅做一致性校验，不是签名边界。
 227:     """
 228: 
 229:     model_config = ConfigDict(extra="forbid")
 230: 
 231:     schema_version: int = IMPORT_ENVELOPE_SCHEMA_VERSION
 232:     target_item_id: str | None = None
 233:     filename: str | None = None
 234:     source_instance_id: str | None = None
 235:     source_document_id: str | None = None
 236:     source_payload_hash: str = Field(min_length=1, max_length=64)
 237:     source_content_hash: str = Field(min_length=1, max_length=64)
 238:     canonicalization_version: int
 239:     space_id: str | None = None
 240:     status: str
 241:     tags: list[str] = Field(default_factory=list)
 242:     extra_metadata: dict[str, Any] = Field(default_factory=dict)
 243:     links: list[dict[str, Any]] = Field(default_factory=list)
 244:     attachments: list[dict[str, Any]] = Field(default_factory=list)
 245:     context_digest: str = ""
 246: 
 247:     def compute_context_digest(self) -> str:
 248:         body = self.model_dump(mode="json", exclude={"context_digest"})
 249:         # envelope 自身已携带 canonicalization_version 字段；把它作为包络版本
 250:         # 传给 canonical 原语，避免与保留键重复（否则 canonicalize 会拒绝）。
 251:         version = int(body.pop("canonicalization_version"))
 252:         return sha256_canonical(body, canonicalization_version=version)
 253: 
 254:     def with_digest(self) -> ImportEnvelope:
 255:         return self.model_copy(update={"context_digest": self.compute_context_digest()})
 256: 
 257:     @classmethod
 258:     def from_review_payload(cls, payload: dict[str, Any]) -> ImportEnvelope | None:
 259:         """从审核单 payload 提取 envelope；无本键返回 None。
 260: 
 261:         结构/摘要不合法抛 ``CanonicalizationError``（调用方 fail-closed）。
 262:         """
 263:         raw = payload.get(IMPORT_ENVELOPE_KEY)
 264:         if raw is None:
 265:             return None
 266:         try:
 267:             envelope = cls.model_validate(raw)
 268:         except Exception as exc:  # pydantic.ValidationError → 稳定失败
 269:             raise CanonicalizationError("导入审核 envelope 结构不合法") from exc
 270:         if envelope.schema_version != IMPORT_ENVELOPE_SCHEMA_VERSION:
 271:             raise CanonicalizationError("导入审核 envelope 版本不受支持")
 272:         expected = envelope.compute_context_digest()
 273:         if envelope.context_digest != expected:
 274:             raise CanonicalizationError("导入审核 envelope 摘要不一致")
 275:         return envelope
 276: 
 277:     def review_payload(self, core_changes: dict[str, Any]) -> dict[str, Any]:
 278:         """构造提交给 Core ReviewService 的 payload（四字段 + envelope）。"""
 279:         body = self.with_digest().model_dump(mode="json")
 280:         return {**core_changes, IMPORT_ENVELOPE_KEY: body}
