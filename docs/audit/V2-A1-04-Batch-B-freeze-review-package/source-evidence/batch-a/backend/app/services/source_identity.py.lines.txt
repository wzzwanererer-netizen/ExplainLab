 215:     """返回 (import_origin_state, 唯一有效身份对, 单记录身份摘要)。"""
 216:     n = len(views)
 217:     if n == 0:
 218:         return "none", None, None
 219:     legal = [view for view in views if not view.flags]
 220:     pairs = {(view.instance, view.document) for view in legal}
 221:     has_duplicate = any(view.duplicate_group_size > 1 for view in views)
 222:     has_conflict = len(pairs) >= 2
 223:     if len(pairs) == 1:

 225:         if db_instance is None or db_instance != instance:
 226:             has_conflict = True

 228:         return "conflict", None, None
 229:     if any("invalid" in view.flags for view in views):
 230:         return "invalid", None, None
 231:     if any("incomplete" in view.flags for view in views):
 232:         return "incomplete", None, None
 233:     if has_duplicate:
 234:         return "duplicate", None, None
 235:     if len(pairs) == 1:
 236:         pair = next(iter(pairs))
 237:         if len(legal) != len(views):
 238:             return "invalid", None, None

 488:     def _publish_archive_no_overwrite(tmp: Path, target: Path, payload: bytes) -> None:

 496:             os.link(tmp, target)

 703:             origins = session.scalar(
 704:                 select(ImportOriginORM.id).where(
 705:                     ImportOriginORM.item_id == entry["item_id"]
 706:                 )
 707:             )
 708:             if origins is not None:
 709:                 raise SourceIdentityError(
 710:                     SOURCE_IDENTITY_INCONSISTENT, "materialized item has origin"
 711:                 )
 712:             if item.source_instance_id != entry["source_instance_id"]:
 713:                 raise SourceIdentityError(
 714:                     SOURCE_IDENTITY_INCONSISTENT, "materialized instance mismatch"
 715:                 )
 716: 
 717:     def _verify_audit(self, record: dict[str, Any], session: Session) -> None:
 718:         if self.audit_digest(session) != record["identity_audit_digest"]:
 719:             raise SourceIdentityError(
 720:                 SOURCE_IDENTITY_INCONSISTENT, "identity audit digest mismatch"
 721:             )
 722: 
 723:     def _commit_and_archive(
 724:         self, record: dict[str, Any], session: Session
 725:     ) -> dict[str, Any]:
 726:         self._verify_audit(record, session)
 727:         committed = dict(record)
 728:         committed["phase"] = "COMMITTED"
 729:         committed["updated_at"] = _now()
 730:         body = {k: v for k, v in committed.items() if k != "journal_checksum"}
 731:         committed["journal_checksum"] = sha256_canonical(body)
 732:         self.journal.write_current(committed)
 733:         # 归档前从 DB 重算（写入 COMMITTED 不改变 DB，但规则要求逐次取证）。
 734:         self._verify_audit(committed, session)
 735:         self.journal.archive_committed(committed)
 736:         self.journal.remove_current()
 737:         return committed
 738: 
 739:     def _recover_materialize(self, record: dict[str, Any]) -> None:
 740:         entry = record["materialization_items"][0]
 741:         session = self._factory()
 742:         try:
 743:             item = session.get(KnowledgeItemORM, entry["item_id"])
 744:             if item is None:
 745:                 raise SourceIdentityError(JOURNAL_CORRUPT, "materialize target missing")
 746:             has_origin = session.scalar(
 747:                 select(ImportOriginORM.id).where(
 748:                     ImportOriginORM.item_id == entry["item_id"]
 749:                 )
 750:             )
 751:             if has_origin is not None:
 752:                 raise SourceIdentityError(
 753:                     SOURCE_IDENTITY_INCONSISTENT, "target has import origin"
 754:                 )
 755:             if item.source_instance_id in (None, ""):
 756:                 stmt = (
 757:                     update(KnowledgeItemORM)
 758:                     .where(
 759:                         KnowledgeItemORM.id == entry["item_id"],
 760:                         or_(
 761:                             KnowledgeItemORM.source_instance_id.is_(None),
 762:                             KnowledgeItemORM.source_instance_id == "",
 763:                         ),
 764:                     )
 765:                     .values(
 766:                         source_instance_id=entry["source_instance_id"],
 767:                         # 物化是身份元数据写，不是内容修改：保持 updated_at 不变，
 768:                         # 否则重复导出 payload_hash 会漂移（D23 稳定导出）。
 769:                         updated_at=KnowledgeItemORM.updated_at,
 770:                     )
 771:                 )
 772:                 if session.execute(stmt).rowcount != 1:
 773:                     session.rollback()
 774:                     raise SourceIdentityError(
 775:                         SOURCE_IDENTITY_INCONSISTENT, "conditional update failed"
 776:                     )
 777:                 session.commit()
 778:                 session.expire_all()
 779:                 item = session.get(KnowledgeItemORM, entry["item_id"])
 780:                 assert item is not None
 781:             if item.source_instance_id != entry["source_instance_id"]:
 782:                 raise SourceIdentityError(
 783:                     SOURCE_IDENTITY_INCONSISTENT, "materialize value conflict"
 784:                 )
 785:             self._verify_record_db(record, session)
 786:             self._verify_audit(record, session)
 787:             self._commit_and_archive(record, session)
 788:         except Exception:
 789:             session.rollback()
 790:             raise
 791:         finally:
 792:             session.close()
 793: 
 794:     def _recover_rotate(self, record: dict[str, Any]) -> None:
 795:         current_id = self._store.read()
 796:         target = record["target_instance_id"]
 797:         old = record["old_instance_id"]
 798:         session = self._factory()
 799:         try:
 800:             self._verify_record_db(record, session)
 801:             if current_id == old:
 802:                 self._verify_audit(record, session)
 803:                 self._store.replace(target)
 804:                 if self._store.read() != target:
 805:                     raise SourceIdentityError(
 806:                         SOURCE_IDENTITY_INCONSISTENT, "rotate id readback"
 807:                     )
 808:                 self._verify_audit(record, session)
 809:             elif current_id != target:
 810:                 raise SourceIdentityError(
 811:                     SOURCE_IDENTITY_INCONSISTENT, "rotate instance mismatch"
 812:                 )
 813:             self._verify_audit(record, session)
 814:             self._commit_and_archive(record, session)
 815:         finally:
 816:             session.close()
 817: 
 818:     def _recover_committed(self, record: dict[str, Any]) -> None:
 819:         session = self._factory()
 820:         try:
 821:             current_id = self._store.read()
 822:             if record["kind"] == "ROTATE":
 823:                 if current_id != record["target_instance_id"]:
 824:                     raise SourceIdentityError(
 825:                         SOURCE_IDENTITY_INCONSISTENT, "committed rotate id mismatch"
 826:                     )
 827:             else:
 828:                 self._verify_record_db(record, session)
 829:             self._verify_audit(record, session)
 830:             self.journal.archive_committed(record)
 831:             self.journal.remove_current()
 832:         finally:
 833:             session.close()

 835:     # -- MATERIALIZE ---------------------------------------------------------
 836:     def ensure_local_export_identity(self, item_id: UUID | str) -> str:
 837:         """导出前确保本地文档有稳定身份：2b 直接复用；2a 走 Journal 物化。"""
 838:         with self._journal_lock:
 839:             self._recover_locked()
 840:             session = self._factory()
 841:             try:
 842:                 classified = self.classify(session, item_id)
 843:             finally:
 844:                 session.close()
 845:             if classified.state == "2b":
 846:                 assert classified.source_instance_id is not None
 847:                 return classified.source_instance_id
 848:             if classified.state == "2a":
 849:                 return self.materialize(item_id).source_instance_id
 850:             if classified.state == "1":
 851:                 assert classified.source_instance_id is not None
 852:                 return classified.source_instance_id
 853:             raise SourceIdentityError(
 854:                 SOURCE_DOCUMENT_UNMATERIALIZED,
 855:                 f"identity state {classified.state} cannot be materialized",
 856:             )
