 415:     """审核批准（同步写入口，D20）。
 416: 
 417:     导入分支：与 submit 共用同一身份键锁；S0 预检 → 获锁 → S1 fresh session
 418:     重读并校验 envelope 未变 → Core approve → 导入钩子补齐 → 单次 commit。
 419:     非导入分支保持既有 K1 行为，不额外解析正文。
 420:     """
 421:     try:
 422:         _s0_payload, s0_digest, lock_key = _preflight_review(review_id)
 423:     except (MarkdownServiceError, SourceIdentityError) as exc:
 424:         raise _import_http_error(exc) from exc

 426:     coordinator: ImportCoordinator = request.app.state.import_coordinator
 427:     identity = getattr(request.app.state, "source_identity", None)
 428:     if identity is None:
 429:         raise HTTPException(status_code=503, detail="identity_service_unavailable")
 430:     context = coordinator.keyed(lock_key) if lock_key else _null_context()
 431:     # P1-02：审核批准的身份写入与 MATERIALIZE/ROTATE 串行；钩子在写前校验
 432:     # 目标条目未被物化，冲突时整事务回滚、审核单保持 PENDING。
 433:     with context, identity.identity_guard():
 434:         session = create_session()

 436:             pending = _pending_review(session, review_id)
 437:             is_import = review_is_import(pending)
 438:             envelope = None

 441:                 if envelope is None or pending.action != ReviewAction.UPDATE_ITEM:
 442:                     raise MarkdownServiceError(
 443:                         IMPORT_ENVELOPE_INVALID,
 444:                         "导入 envelope 非法",
 445:                         status_code=409,
 446:                     )
 447:             if s0_digest is not None and (
 448:                 not is_import
 449:                 or s0_digest != _payload_digest(pending.payload).encode("utf-8")
 450:             ):
 451:                 raise MarkdownServiceError(
 452:                     IMPORT_ENVELOPE_INVALID,
 453:                     "审核 envelope 在获锁前被改动",
