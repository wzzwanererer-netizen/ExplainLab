 120: def _collect_documents(
 121:     body: MarkdownImportIn,
 122: ) -> tuple[list[MarkdownImportDocument], dict[str, Any] | None]:
 123:     manifest = body.manifest
 124:     documents: list[MarkdownImportDocument] = []
 125:     if body.archive_base64:
 126:         documents, archive_manifest = _from_archive(body.archive_base64)
 127:         if archive_manifest is not None:
 128:             manifest = archive_manifest
 129:     elif body.documents is not None:
 130:         documents = [
 131:             _to_document(
 132:                 content=doc.content,
 133:                 filename=doc.filename,
 134:                 target_item_id=doc.target_item_id,
 135:                 frontmatter_resolution=doc.frontmatter_resolution,
 136:                 attachments_requested=bool(doc.attachments),
 137:             )
 138:             for doc in body.documents
 139:         ]
 140:     elif body.content is not None:
 141:         documents = [
 142:             _to_document(
 143:                 content=body.content,
 144:                 filename=body.filename,
 145:                 target_item_id=body.target_item_id,
 146:                 frontmatter_resolution=body.frontmatter_resolution,
 147:                 attachments_requested=bool(body.attachments),
 148:             )
 149:         ]
 150:     if not documents:
 151:         raise MarkdownServiceError("IMPORT_EMPTY", "no documents")
 152:     return documents, manifest

 155: def _sync_indexes(session_factory, index_ids: list[UUID], embedder: Embedder | None) -> str:
 156:     if not index_ids:
 157:         return "ok"
 158:     session: Session = session_factory()
 159:     try:
 160:         asyncio.run(sync_index(session, embedder, index_ids=index_ids))
 161:         session.commit()
 162:         return "ok"
 163:     except Exception as exc:  # noqa: BLE001 - 提交后同步失败只告警，不回滚业务数据
 164:         session.rollback()
 165:         logger.warning("导入后索引同步失败: %s", type(exc).__name__)
 166:         return "failed"
 167:     finally:
 168:         session.close()

 171: def _import_response(
 172:     service: MarkdownService,
 173:     documents: list[MarkdownImportDocument],
 174:     *,
 175:     space_id: str | None,
 176:     manifest: dict[str, Any] | None,
 177:     batch_id: str | None,
 178:     embedder: Embedder | None,
 179:     reject_attachments: bool,
 180:     response: Response,
 181:     single: bool,
 182: ) -> dict[str, Any]:
 183:     try:
 184:         outcome = service.import_documents(
 185:             documents,
 186:             space_id=space_id,
 187:             manifest=manifest,
 188:             batch_id=batch_id,
 189:             reject_attachments=reject_attachments,
 190:         )
 191:     except (MarkdownServiceError, SourceIdentityError, ExporterMetadataError) as exc:
 192:         raise _http_error(exc) from exc
 193:     index_state = _sync_indexes(service.session_factory, outcome.index_ids, embedder)
 194:     body = outcome.report.model_dump(mode="json")
 195:     body["index_sync"] = index_state
 196:     if single and outcome.report.items:
 197:         first = outcome.report.items[0]
 198:         if first.status.value == "error" and IMPORT_LOCK_TIMEOUT in first.errors:
 199:             raise HTTPException(status_code=409, detail=IMPORT_LOCK_TIMEOUT)
 200:         if first.status.value == "pending_review":
 201:             response.status_code = 202
 202:         elif first.status.value == "created":
 203:             response.status_code = 201
 204:     return body
